=== query and headers
import axios from 'axios';

const response = await axios.request({
  method: 'get',
  url: 'https://api.example.com/v1/items?q=red%20shoes&page=2',
  headers: {
    'Accept': 'application/json',
    'X-Trace': 'it\'s "quoted", second',
  },
  responseType: 'text',
  validateStatus: () => true,
});

console.log(response.status);
console.log(response.data);

=== insecure, no redirects, redirect limit ignored
import axios from 'axios';
import https from 'node:https';

const response = await axios.request({
  method: 'get',
  url: 'https://self-signed.example.com/health',
  maxRedirects: 0,
  timeout: 1500,
  httpsAgent: new https.Agent({ rejectUnauthorized: false }),
  responseType: 'text',
  validateStatus: () => true,
});

console.log(response.status);
console.log(response.data);

=== soap envelope
import axios from 'axios';

const body = `<soap:Envelope xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/">
  <soap:Body><Add><a>1</a><b>2</b></Add></soap:Body>
</soap:Envelope>`;

const response = await axios.request({
  method: 'post',
  url: 'http://soap.example.com/calculator',
  headers: {
    'Content-Type': 'text/xml;charset=UTF-8',
    'SOAPAction': '"http://tempuri.org/Add"',
  },
  data: body,
  maxRedirects: 0,
  responseType: 'text',
  validateStatus: () => true,
});

console.log(response.status);
console.log(response.data);
