=== query and headers
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.Response;

public class Main {
    public static void main(String[] args) throws Exception {
        OkHttpClient client = new OkHttpClient.Builder()
                .build();
        Request request = new Request.Builder()
                .url("https://api.example.com/v1/items?q=red%20shoes&page=2")
                .method("GET", null)
                .addHeader("Accept", "application/json")
                .addHeader("X-Trace", "it's \"quoted\"")
                .addHeader("X-Trace", "second")
                .build();
        try (Response response = client.newCall(request).execute()) {
            System.out.println(response.code());
            System.out.println(response.body().string());
        }
    }
}

=== multipart body
import java.io.File;
import okhttp3.MediaType;
import okhttp3.MultipartBody;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.RequestBody;
import okhttp3.Response;

public class Main {
    public static void main(String[] args) throws Exception {
        OkHttpClient client = new OkHttpClient.Builder()
                .build();
        RequestBody body = new MultipartBody.Builder()
                .setType(MultipartBody.FORM)
                .addFormDataPart("title", "Report")
                .addFormDataPart("file", "report.pdf", RequestBody.create(new File("/tmp/report.pdf"), MediaType.parse("application/pdf")))
                .build();
        Request request = new Request.Builder()
                .url("https://api.example.com/upload")
                .method("POST", body)
                .build();
        try (Response response = client.newCall(request).execute()) {
            System.out.println(response.code());
            System.out.println(response.body().string());
        }
    }
}

=== soap envelope
import okhttp3.MediaType;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.RequestBody;
import okhttp3.Response;

public class Main {
    public static void main(String[] args) throws Exception {
        OkHttpClient client = new OkHttpClient.Builder()
                .followRedirects(false)
                .followSslRedirects(false)
                .build();
        String content = """
<soap:Envelope xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/">
  <soap:Body><Add><a>1</a><b>2</b></Add></soap:Body>
</soap:Envelope>""";
        RequestBody body = RequestBody.create(content, MediaType.parse("text/xml;charset=UTF-8"));
        Request request = new Request.Builder()
                .url("http://soap.example.com/calculator")
                .method("POST", body)
                .addHeader("SOAPAction", "\"http://tempuri.org/Add\"")
                .build();
        try (Response response = client.newCall(request).execute()) {
            System.out.println(response.code());
            System.out.println(response.body().string());
        }
    }
}
