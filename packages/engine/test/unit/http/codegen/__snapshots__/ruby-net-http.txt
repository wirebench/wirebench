# Note: This request follows redirects; the library does not, so the snippet stops at the first.
require 'net/http'
require 'uri'

uri = URI('https://api.example.com/v1/items?q=red%20shoes&page=2')
request = Net::HTTP::Get.new(uri)
request['Accept'] = 'application/json'
request['X-Trace'] = 'it\'s "quoted"'
request.add_field('X-Trace', 'second')

response = Net::HTTP.start(
  uri.hostname,
  uri.port,
  use_ssl: uri.scheme == 'https'
) do |http|
  http.request(request)
end

puts response.code
