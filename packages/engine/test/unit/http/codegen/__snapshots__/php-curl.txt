=== query and headers
<?php

$curl = curl_init();
curl_setopt_array($curl, [
    CURLOPT_URL => 'https://api.example.com/v1/items?q=red%20shoes&page=2',
    CURLOPT_CUSTOMREQUEST => 'GET',
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_FOLLOWLOCATION => true,
    CURLOPT_HTTPHEADER => [
        'Accept: application/json',
        'X-Trace: it\'s "quoted"',
        'X-Trace: second',
    ],
]);

$response = curl_exec($curl);
if ($response === false) {
    fwrite(STDERR, curl_error($curl) . PHP_EOL);
    exit(1);
}
echo curl_getinfo($curl, CURLINFO_RESPONSE_CODE) . PHP_EOL;
echo $response . PHP_EOL;
curl_close($curl);

  "price": 9.5
}';

=== soap envelope
<?php

$body = '<soap:Envelope xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/">
  <soap:Body><Add><a>1</a><b>2</b></Add></soap:Body>
</soap:Envelope>';

$curl = curl_init();
curl_setopt_array($curl, [
    CURLOPT_URL => 'http://soap.example.com/calculator',
    CURLOPT_CUSTOMREQUEST => 'POST',
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_HTTPHEADER => [
        'Content-Type: text/xml;charset=UTF-8',
        'SOAPAction: "http://tempuri.org/Add"',
    ],
    CURLOPT_POSTFIELDS => $body,
]);

$response = curl_exec($curl);
if ($response === false) {
    fwrite(STDERR, curl_error($curl) . PHP_EOL);
    exit(1);
}
echo curl_getinfo($curl, CURLINFO_RESPONSE_CODE) . PHP_EOL;
echo $response . PHP_EOL;
curl_close($curl);
