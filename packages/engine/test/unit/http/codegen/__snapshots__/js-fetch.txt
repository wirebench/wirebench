=== query and headers
const response = await fetch('https://api.example.com/v1/items?q=red%20shoes&page=2', {
  method: 'GET',
  headers: {
    'Accept': 'application/json',
    'X-Trace': 'it\'s "quoted", second',
  },
  redirect: 'follow',
});

console.log(response.status);
console.log(await response.text());

=== json body
const body = `{
  "name": "Widget",
  "price": 9.5
}`;

const response = await fetch('https://api.example.com/v1/items', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
  },
  body,
  redirect: 'follow',
  signal: AbortSignal.timeout(30000),
});

console.log(response.status);
console.log(await response.text());

=== multipart body
import { readFile } from 'node:fs/promises';

const body = new FormData();
body.append('title', 'Report');
body.append('file', new Blob([await readFile('/tmp/report.pdf')], { type: 'application/pdf' }), 'report.pdf');

const response = await fetch('https://api.example.com/upload', {
  method: 'POST',
  body,
  redirect: 'follow',
});

console.log(response.status);
console.log(await response.text());
